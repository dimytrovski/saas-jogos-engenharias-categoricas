  id: string;

export function getRevenueTargetBrl(): 30000 {
  return 30000;
}

export function formatCurrencyBrl(value: number): string {
  return new Intl.NumberFormat("pt-BR", {
    style: "currency",
    currency: "BRL",
    minimumFractionDigits: 2,
    maximumFractionDigits: 2
  }).format(value);
}

export function calculateRevenueTotalBrl(
  deploymentsCount: number,
  deploymentUnitBrl: number = 10000,
  passesCount: number = 0,
  passUnitBrl: number = 0
): number {
  const safeDeployments = Math.max(0, Number(deploymentsCount) || 0);
  const safeDepUnit = Math.max(0, Number(deploymentUnitBrl) || 0);
  const safePasses = Math.max(0, Number(passesCount) || 0);
  const safePassUnit = Math.max(0, Number(passUnitBrl) || 0);
  return Number((safeDeployments * safeDepUnit + safePasses * safePassUnit).toFixed(2));
}

interface FlowParticle {
  x: number;
  y: number;
  px: number;
  py: number;
  vx: number;
  vy: number;
  life: number;
  maxLife: number;
  alpha: number;

function FlowFieldCanvas({
  rgb,
  speedMultiplier
}: {
  rgb: string;
  speedMultiplier: number;
}) {
  const canvasRef = useRef<HTMLCanvasElement | null>(null);
  const themeRef = useRef({ rgb, speedMultiplier });
  const pointerRef = useRef({ x: -9999, y: -9999, active: false });

  useEffect(() => {
    themeRef.current = { rgb, speedMultiplier };
  }, [rgb, speedMultiplier]);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext("2d", { alpha: true });
    if (!ctx) return;

    const reducedMotion =
      typeof window !== "undefined" &&
      window.matchMedia &&
      window.matchMedia("(prefers-reduced-motion: reduce)").matches;

    let width = window.innerWidth;
    let height = window.innerHeight;
    let dpr = Math.min(window.devicePixelRatio || 1, 2);
    let rafId = 0;
    let tick = 0;
    let isVisible = document.visibilityState === "visible";

    const particleCount = width < 640 ? 48 : 96;
    const particles: FlowParticle[] = [];

    const spawnParticle = (initial = false): FlowParticle => {
      const x = Math.random() * width;
      const y = Math.random() * height;
      const maxLife = 90 + Math.random() * 140;
      return {
        x,
        y,
        px: x,
        py: y,
        vx: 0,
        vy: 0,
        life: initial ? Math.random() * maxLife : 0,
        maxLife,
        alpha: 0.08 + Math.random() * 0.16
      };
    };

    const resize = () => {
      width = window.innerWidth;
      height = window.innerHeight;
      dpr = Math.min(window.devicePixelRatio || 1, 2);
      canvas.width = Math.floor(width * dpr);
      canvas.height = Math.floor(height * dpr);
      canvas.style.width = `${width}px`;
      canvas.style.height = `${height}px`;
      ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    };

    resize();
    for (let i = 0; i < particleCount; i++) {
      particles.push(spawnParticle(true));
    }

    const onPointerMove = (e: PointerEvent) => {
      pointerRef.current = { x: e.clientX, y: e.clientY, active: true };
    };

    const onPointerLeave = () => {
      pointerRef.current.active = false;
    };

    const onVisibilityChange = () => {
      isVisible = document.visibilityState === "visible";
      if (isVisible && !reducedMotion && !rafId) {
        rafId = window.requestAnimationFrame(render);
      }
    };

    const renderFrame = (stepOnce: boolean) => {
      tick += 0.0045 * themeRef.current.speedMultiplier;
      ctx.clearRect(0, 0, width, height);

      const { rgb: activeRgb, speedMultiplier } = themeRef.current;
      const ptr = pointerRef.current;

      for (let i = 0; i < particles.length; i++) {
        const p = particles[i];
        p.px = p.x;
        p.py = p.y;

        const nx = p.x * 0.0024;
        const ny = p.y * 0.0024;
        const angle =
          Math.sin(nx * 2.1 + tick) * Math.cos(ny * 2.1 - tick * 0.8) * Math.PI * 1.45 +
          Math.sin((nx + ny) * 1.3 + tick * 0.5) * 0.65;

        let forceX = Math.cos(angle) * 0.85 * speedMultiplier;
        let forceY = Math.sin(angle) * 0.85 * speedMultiplier;

        if (ptr.active) {
          const dx = ptr.x - p.x;
          const dy = ptr.y - p.y;
          const distSq = dx * dx + dy * dy;
          const radius = 190;
          if (distSq > 1 && distSq < radius * radius) {
            const dist = Math.sqrt(distSq);
            const pull = (1 - dist / radius) * 0.55;
            const swirlX = -dy / dist;
            const swirlY = dx / dist;
            forceX += (dx / dist) * pull * 0.45 + swirlX * pull * 0.85;
            forceY += (dy / dist) * pull * 0.45 + swirlY * pull * 0.85;
          }
        }

        p.vx = p.vx * 0.86 + forceX * 0.14;
        p.vy = p.vy * 0.86 + forceY * 0.14;
        p.x += p.vx;
        p.y += p.vy;
        p.life += stepOnce ? 1 : 0.6;

        const fade = Math.sin(Math.min(p.life / p.maxLife, 1) * Math.PI);
        const strokeAlpha = Math.max(0, p.alpha * fade);

        ctx.beginPath();
        ctx.moveTo(p.px, p.py);
        ctx.lineTo(p.x, p.y);
        ctx.strokeStyle = `rgba(${activeRgb}, ${strokeAlpha.toFixed(3)})`;
        ctx.lineWidth = 1.25;
        ctx.stroke();

        if (
          p.life >= p.maxLife ||
          p.x < -30 ||
          p.x > width + 30 ||
          p.y < -30 ||
          p.y > height + 30
        ) {
          particles[i] = spawnParticle(false);
        }
      }
    };

    const render = () => {
      rafId = 0;
      if (!isVisible) return;
      renderFrame(true);
      rafId = window.requestAnimationFrame(render);
    };

    renderFrame(false);
    if (!reducedMotion && isVisible) {
      rafId = window.requestAnimationFrame(render);
    }

    window.addEventListener("resize", resize);
    window.addEventListener("pointermove", onPointerMove, { passive: true });
    window.addEventListener("pointerleave", onPointerLeave);
    document.addEventListener("visibilitychange", onVisibilityChange);

    return () => {
      if (rafId) window.cancelAnimationFrame(rafId);
      window.removeEventListener("resize", resize);
      window.removeEventListener("pointermove", onPointerMove);
      window.removeEventListener("pointerleave", onPointerLeave);
      document.removeEventListener("visibilitychange", onVisibilityChange);
    };
  }, []);

  return (
    <canvas
      ref={canvasRef}
      aria-hidden="true"
      className="pointer-events-none fixed inset-0 z-0"
    />
  );
}

          })}
        </div>

          <span className="font-mono text-xs text-slate-400">
            {current.code} · {activeScenario.label} · {activeScenario.formattedTotalBrl}

            <span>{current.code}</span>
            <ArrowUpRight className="h-4 w-4" />

}
