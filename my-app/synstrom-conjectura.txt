const http = require("http");
const pool = require("./db");

const PORT = 3000;

function readJsonBody(req) {
  return new Promise((resolve, reject) => {
    let body = "";
    req.on("data", (chunk) => {
      body += chunk.toString();
    });
    req.on("end", () => {
      if (!body) return resolve({});
      try {
        resolve(JSON.parse(body));
      } catch (err) {
        reject(new Error("JSON inválido"));
      }
    });
    req.on("error", reject);
  });
}

const server = http.createServer(async (req, res) => {
  res.setHeader("Content-Type", "application/json; charset=utf-8");

  if (req.method === "GET" && req.url === "/health") {
    res.writeHead(200);
    return res.end(JSON.stringify({ status: "ok" }));
  }

  if (req.method === "GET" && req.url === "/users") {
    try {
      const [rows] = await pool.execute(
        "SELECT id, email, plan_status, created_at FROM users ORDER BY id DESC"
      );
      res.writeHead(200);
      return res.end(JSON.stringify({ users: rows }));
    } catch (err) {
      res.writeHead(500);
      return res.end(JSON.stringify({ error: err.message }));
    }
  }

  if (req.method === "POST" && req.url === "/users") {
    try {
      const { email, password_hash } = await readJsonBody(req);
      if (!email || !password_hash) {
        res.writeHead(400);
        return res.end(
          JSON.stringify({ error: "Os campos email e password_hash são obrigatórios." })
        );
      }

      const [result] = await pool.execute(
        "INSERT INTO users (email, password_hash) VALUES (?, ?)",
        [email, password_hash]
      );

      res.writeHead(201);
      return res.end(
        JSON.stringify({
          id: result.insertId,
          email,
          plan_status: "free",
        })
      );
    } catch (err) {
      if (err.code === "ER_DUP_ENTRY") {
        res.writeHead(409);
        return res.end(JSON.stringify({ error: "Esse e-mail já está registado." }));
      }
      res.writeHead(500);
      return res.end(JSON.stringify({ error: err.message }));
    }
  }

  res.writeHead(404);
  res.end(JSON.stringify({ error: "Rota não encontrada." }));
});

server.listen(PORT, () => {
  console.log(`Servidor a correr em http://localhost:${PORT}`);
});

"use client";

import React, { createContext, useContext, useState, ReactNode } from 'react';

interface TelemetryData {
  ping: number;
  jitter: number;
  packetLoss: number;
  isHydrated: boolean;
}

interface TelemetryContextType {
  telemetry: TelemetryData;
  setTelemetry: React.Dispatch<React.SetStateAction<TelemetryData>>;
  markHydrated: () => void;
}

const TelemetryContext = createContext<TelemetryContextType | undefined>(undefined);

export const TelemetryProvider = ({ children }: { children: ReactNode }) => {
  const [telemetry, setTelemetry] = useState<TelemetryData>({
    ping: 0,
    jitter: 0,
    packetLoss: 0,
    isHydrated: false,
  });

  const markHydrated = () => {
    setTelemetry(prev => ({ ...prev, isHydrated: true }));
  };

  return (
    <TelemetryContext.Provider value={{ telemetry, setTelemetry, markHydrated }}>
      {children}
    </TelemetryContext.Provider>
  );
};

export const useTelemetry = () => {
  const context = useContext(TelemetryContext);
  if (!context) {
    throw new Error('useTelemetry must be used within a TelemetryProvider');
  }
  return context;
};

interface FlowParticle {
  x: number;
  y: number;
  px: number;
  py: number;
  vx: number;
  vy: number;
  life: number;
  maxLife: number;
  alpha: number;
}

function FlowFieldCanvas({
  rgb,
  speedMultiplier
}: {
  rgb: string;
  speedMultiplier: number;
}) {
  const canvasRef = useRef<HTMLCanvasElement | null>(null);
  const themeRef = useRef({ rgb, speedMultiplier });
  const pointerRef = useRef({ x: -9999, y: -9999, active: false });

  useEffect(() => {
    themeRef.current = { rgb, speedMultiplier };
  }, [rgb, speedMultiplier]);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext("2d", { alpha: true });
    if (!ctx) return;

    const reducedMotion =
      typeof window !== "undefined" &&
      window.matchMedia &&
      window.matchMedia("(prefers-reduced-motion: reduce)").matches;

    let width = window.innerWidth;
    let height = window.innerHeight;
    let dpr = Math.min(window.devicePixelRatio || 1, 2);
    let rafId = 0;
    let tick = 0;
    let isVisible = document.visibilityState === "visible";

    const particleCount = width < 640 ? 48 : 96;
    const particles: FlowParticle[] = [];

    const spawnParticle = (initial = false): FlowParticle => {
      const x = Math.random() * width;
      const y = Math.random() * height;
      const maxLife = 90 + Math.random() * 140;
      return {
        x,
        y,
        px: x,
        py: y,
        vx: 0,
        vy: 0,
        life: initial ? Math.random() * maxLife : 0,
        maxLife,
        alpha: 0.08 + Math.random() * 0.16
      };
    };

    const resize = () => {
      width = window.innerWidth;
      height = window.innerHeight;
      dpr = Math.min(window.devicePixelRatio || 1, 2);
      canvas.width = Math.floor(width * dpr);
      canvas.height = Math.floor(height * dpr);
      canvas.style.width = `${width}px`;
      canvas.style.height = `${height}px`;
      ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    };

    resize();
    for (let i = 0; i < particleCount; i++) {
      particles.push(spawnParticle(true));
    }

    const onPointerMove = (e: PointerEvent) => {
      pointerRef.current = { x: e.clientX, y: e.clientY, active: true };
    };

    const onPointerLeave = () => {
      pointerRef.current.active = false;
    };

    const onVisibilityChange = () => {
      isVisible = document.visibilityState === "visible";
      if (isVisible && !reducedMotion && !rafId) {
        rafId = window.requestAnimationFrame(render);
      }
    };

    const renderFrame = (stepOnce: boolean) => {
      tick += 0.0045 * themeRef.current.speedMultiplier;
      ctx.clearRect(0, 0, width, height);

      const { rgb: activeRgb, speedMultiplier } = themeRef.current;
      const ptr = pointerRef.current;

      for (let i = 0; i < particles.length; i++) {
        const p = particles[i];
        p.px = p.x;
        p.py = p.y;

        const nx = p.x * 0.0024;
        const ny = p.y * 0.0024;
        const angle =
          Math.sin(nx * 2.1 + tick) * Math.cos(ny * 2.1 - tick * 0.8) * Math.PI * 1.45 +
          Math.sin((nx + ny) * 1.3 + tick * 0.5) * 0.65;

        let forceX = Math.cos(angle) * 0.85 * speedMultiplier;
        let forceY = Math.sin(angle) * 0.85 * speedMultiplier;

        if (ptr.active) {
          const dx = ptr.x - p.x;
          const dy = ptr.y - p.y;
          const distSq = dx * dx + dy * dy;
          const radius = 190;
          if (distSq > 1 && distSq < radius * radius) {
            const dist = Math.sqrt(distSq);
            const pull = (1 - dist / radius) * 0.55;
            const swirlX = -dy / dist;
            const swirlY = dx / dist;
            forceX += (dx / dist) * pull * 0.45 + swirlX * pull * 0.85;
            forceY += (dy / dist) * pull * 0.45 + swirlY * pull * 0.85;
          }
        }

        p.vx = p.vx * 0.86 + forceX * 0.14;
        p.vy = p.vy * 0.86 + forceY * 0.14;
        p.x += p.vx;
        p.y += p.vy;
        p.life += stepOnce ? 1 : 0.6;

        const fade = Math.sin(Math.min(p.life / p.maxLife, 1) * Math.PI);
        const strokeAlpha = Math.max(0, p.alpha * fade);

        ctx.beginPath();
        ctx.moveTo(p.px, p.py);
        ctx.lineTo(p.x, p.y);
        ctx.strokeStyle = `rgba(${activeRgb}, ${strokeAlpha.toFixed(3)})`;
        ctx.lineWidth = 1.25;
        ctx.stroke();

        if (
          p.life >= p.maxLife ||
          p.x < -30 ||
          p.x > width + 30 ||
          p.y < -30 ||
          p.y > height + 30
        ) {
          particles[i] = spawnParticle(false);
        }
      }
    };

    const render = () => {
      rafId = 0;
      if (!isVisible) return;
      renderFrame(true);
      rafId = window.requestAnimationFrame(render);
    };

    renderFrame(false);
    if (!reducedMotion && isVisible) {
      rafId = window.requestAnimationFrame(render);
    }

    window.addEventListener("resize", resize);
    window.addEventListener("pointermove", onPointerMove, { passive: true });
    window.addEventListener("pointerleave", onPointerLeave);
    document.addEventListener("visibilitychange", onVisibilityChange);

    return () => {
      if (rafId) window.cancelAnimationFrame(rafId);
      window.removeEventListener("resize", resize);
      window.removeEventListener("pointermove", onPointerMove);
      window.removeEventListener("pointerleave", onPointerLeave);
      document.removeEventListener("visibilitychange", onVisibilityChange);
    };
  }, []);

  return (
    <canvas
      ref={canvasRef}
      aria-hidden="true"
      className="pointer-events-none fixed inset-0 z-0"
    />
  );
}

import React, { createContext, useContext, useState, ReactNode } from 'react';

interface TelemetryData {
  ping: number;
  jitter: number;
  packetLoss: number;
  isHydrated: boolean;
}

interface TelemetryContextType {
  telemetry: TelemetryData;
  setTelemetry: React.Dispatch<React.SetStateAction<TelemetryData>>;
  markHydrated: () => void;
}

const TelemetryContext = createContext<TelemetryContextType | undefined>(undefined);

export const TelemetryProvider = ({ children }: { children: ReactNode }) => {
  const [telemetry, setTelemetry] = useState<TelemetryData>({
    ping: 0,
    jitter: 0,
    packetLoss: 0,
    isHydrated: false,
  });

  const markHydrated = () => {
    setTelemetry(prev => ({ ...prev, isHydrated: true }));
  };

  return (
    <TelemetryContext.Provider value={{ telemetry, setTelemetry, markHydrated }}>
      {children}
    </TelemetryContext.Provider>
  );
};

export const useTelemetry = () => {
  const context = useContext(TelemetryContext);
  if (!context) {
    throw new Error('useTelemetry must be used within a TelemetryProvider');
  }
  return context;
};

      params.push(formatNowSql());
    }
    if (playerId) {
      whereParts.push("player_id = ?");
      params.push(playerId);
    }

    const sql = `SELECT * FROM player_digest_schedules WHERE ${whereParts.join(" AND ")} ORDER BY scheduled_for ASC LIMIT 50`;
    const [rows] = await pool.query<DbScheduleRow[]>(sql, params);

    const origin = req.nextUrl.origin || "http://localhost:3000";
    const results: Array<{
      id: number;
      playerId: string;
      to: string;
      deliveryMode: "smtp" | "simulated";
      messageId?: string;
      nextStatus: string;
      nextScheduledFor?: string;
    }> = [];

    for (const row of rows) {
      const freq = normalizeFrequency(row.frequency_mode, "weekly");
      const prefHour = normalizeHour(row.preferred_hour, 20);
      const record: DigestPlayerRecord = {
        id: Number(row.id),
        playerId: row.player_id,
        playerName: row.player_name,
        playerEmail: row.player_email,
        identityRole: row.identity_role,
        identityBio: row.identity_bio || "",
        brandLogoUrl: row.brand_logo_url,
        brandName: row.brand_name,
        timezone: row.timezone,
        scheduledFor: String(row.scheduled_for),
        frequencyMode: freq,
        preferredHour: prefHour,
        focusGoal: row.focus_goal,
        progressScore: normalizeScore(row.progress_score, 70),
        progressSummary: row.progress_summary || "",
        sendByDefault: Boolean(row.send_by_default)
      };

      const acceptUrl = `${origin}/api/progress-digest/adapt?scheduleId=${row.id}&playerId=${encodeURIComponent(row.player_id)}`;
      const emailPayload = buildProgressEmailPayload(record, acceptUrl);
      const delivery = await sendDigestEmail({
        to: emailPayload.to,
        subject: emailPayload.subject,
        html: emailPayload.html,
        text: emailPayload.text
      });

      const nowSql = formatNowSql();

      if (freq === "once") {
        await pool.query(
          "UPDATE player_digest_schedules SET status = 'sent', last_sent_at = ? WHERE id = ?",
          [nowSql, row.id]
        );
        results.push({
          id: Number(row.id),
          playerId: row.player_id,
          to: row.player_email,
          deliveryMode: delivery.mode,
          messageId: delivery.messageId,
          nextStatus: "sent"
        });
      } else {
        const nextSql = advanceSqlDateTime(String(row.scheduled_for), freq, prefHour);
        await pool.query(
          "UPDATE player_digest_schedules SET last_sent_at = ?, scheduled_for = ?, status = 'pending' WHERE id = ?",
          [nowSql, nextSql, row.id]
        );
        results.push({
          id: Number(row.id),
          playerId: row.player_id,
          to: row.player_email,
          deliveryMode: delivery.mode,
          messageId: delivery.messageId,
          nextStatus: "pending",
          nextScheduledFor: nextSql
        });
      }
    }

    return NextResponse.json({
      ok: true,
      source: "mariadb",
      forceAll,
      processedCount: results.length,
      results
    });
  } catch (err) {
    const msg = err instanceof Error ? err.message : "Erro ao executar disparo de progress digest";
    return NextResponse.json({ ok: false, error: msg }, { status: 500 });
  }
}

import { NextRequest, NextResponse } from "next/server";
import { RowDataPacket } from "mysql2/promise";
import { getDigestPool } from "@/lib/digest-db";

interface TableNameRow extends RowDataPacket {
  TABLE_NAME: string;
}

interface ColumnInfoRow extends RowDataPacket {
  COLUMN_NAME: string;
}

interface GenericRow extends RowDataPacket {
  [key: string]: unknown;
}

const SENSITIVE_COLUMNS = new Set([
  "password_hash",
  "password",
  "senha",
  "token",
  "secret",
  "api_key",
  "salt"
]);

function sanitizeRow(row: Record<string, unknown>): Record<string, unknown> {
  const clean: Record<string, unknown> = {};
  for (const [k, v] of Object.entries(row)) {
    if (!SENSITIVE_COLUMNS.has(k.toLowerCase())) {
      clean[k] = v;
    }
  }
  return clean;
}

function pickField(row: Record<string, unknown>, candidates: string[], fallback = ""): string {
  for (const key of candidates) {
    const exact = row[key];
    if (exact !== undefined && exact !== null && String(exact).trim() !== "") {
      return String(exact).trim();
    }
    const lowerKey = Object.keys(row).find((k) => k.toLowerCase() === key.toLowerCase());
    if (lowerKey && row[lowerKey] !== undefined && row[lowerKey] !== null && String(row[lowerKey]).trim() !== "") {
      return String(row[lowerKey]).trim();
    }
  }
  return fallback;
}

export async function GET(req: NextRequest) {
  try {
    const requestedTable = req.nextUrl.searchParams.get("table")?.trim() || "";
    const requestedPlayerId = req.nextUrl.searchParams.get("playerId")?.trim() || "";

    const pool = await getDigestPool();
    const [dbNameRows] = await pool.query<GenericRow[]>("SELECT DATABASE() AS current_db");
    const currentDb = String(dbNameRows?.[0]?.current_db || "meu_saas");

    const [tableRows] = await pool.query<TableNameRow[]>(
      `SELECT TABLE_NAME
       FROM information_schema.TABLES
       WHERE TABLE_SCHEMA = DATABASE()
       ORDER BY TABLE_NAME ASC`
    );
    const availableTables = tableRows.map((r) => String(r.TABLE_NAME));

    let streamsCount = 0;
    let licensesCount = 0;
    if (availableTables.includes("streams")) {
      const [sRows] = await pool.query<GenericRow[]>("SELECT COUNT(*) AS total FROM `streams`");
      streamsCount = Number(sRows?.[0]?.total || 0);
    }
    if (availableTables.includes("stream_licenses")) {
      const [lRows] = await pool.query<GenericRow[]>("SELECT COUNT(*) AS total FROM `stream_licenses`");
      licensesCount = Number(lRows?.[0]?.total || 0);
    }

    const priorityTables = [
      requestedTable,
      "users",
      "players",
      "jogadores",
      "usuarios",
      "streams",
      "stream_licenses",
      "player_digest_schedules"
    ].filter((t) => t && availableTables.includes(t));

    const chosenTable = priorityTables[0] || availableTables[0] || "";
    if (!chosenTable) {
      return NextResponse.json({
        ok: true,
        database: currentDb,
        tables: [],
        records: [],
        metrics: { streamsCount, licensesCount }
      });
    }

    const [colRows] = await pool.query<ColumnInfoRow[]>(
      `SELECT COLUMN_NAME
       FROM information_schema.COLUMNS
       WHERE TABLE_SCHEMA = DATABASE() AND TABLE_NAME = ?
       ORDER BY ORDINAL_POSITION ASC`,
      [chosenTable]
    );
    const columns = colRows
      .map((c) => String(c.COLUMN_NAME))
      .filter((c) => !SENSITIVE_COLUMNS.has(c.toLowerCase()));

    let querySql = `SELECT * FROM \`${chosenTable}\` ORDER BY 1 DESC LIMIT 25`;
    const queryParams: unknown[] = [];

    if (requestedPlayerId) {
      const idCol = columns.find((c) => ["player_id", "playerid", "user_id", "userid", "id"].includes(c.toLowerCase()));
      if (idCol) {
        querySql = `SELECT * FROM \`${chosenTable}\` WHERE \`${idCol}\` LIKE ? ORDER BY 1 DESC LIMIT 25`;
        queryParams.push(`%${requestedPlayerId}%`);
      }
    }

    const [dataRows] = await pool.query<GenericRow[]>(querySql, queryParams);

    const mappedProfiles = dataRows.map((row, idx) => {
      const safeRaw = sanitizeRow(row as Record<string, unknown>);
      const idVal = pickField(safeRaw, ["player_id", "playerId", "user_id", "userId", "id", "codigo"], `${idx + 1}`);
      const emailVal = pickField(safeRaw, ["player_email", "playerEmail", "email", "mail"], "streamer@exemplo.com");
      const defaultName = emailVal.includes("@") ? emailVal.split("@")[0] : `Jogador ${idVal}`;
      const nameVal = pickField(safeRaw, ["player_name", "playerName", "name", "nome", "username", "apelido"], defaultName);
      const planStatus = pickField(safeRaw, ["plan_status", "identity_role", "role", "status"], "active");
      const roleVal = `Criador / Plano ${planStatus}`;

      const computedScore = Math.min(
        100,
        Math.max(25, 50 + (planStatus.toLowerCase() === "active" ? 15 : 0) + streamsCount * 5 + licensesCount * 5)
      );

      const summaryVal =
        `Conta ${emailVal} (${planStatus}) na base ${currentDb}.${chosenTable} | Streams: ${streamsCount} | Licencas: ${licensesCount}.`;

      return {
        raw: safeRaw,
        profile: {
          playerId: idVal,
          playerName: nameVal,
          playerEmail: emailVal,
          identityRole: roleVal,
          identityBio: `Perfil ativo em ${currentDb} com foco em transmissao, licenciamento e evolucao continua.`,
          progressScore: computedScore,
          progressSummary: summaryVal
        }
      };
    });

    return NextResponse.json({
      ok: true,
      database: currentDb,
      table: chosenTable,
      tables: availableTables,
      columns,
      metrics: { streamsCount, licensesCount },
      total: mappedProfiles.length,
      records: mappedProfiles
    });
  } catch (err) {
    const msg = err instanceof Error ? err.message : "Erro ao consultar dados reais do HeidiSQL";
    return NextResponse.json({ ok: false, error: msg }, { status: 500 });
  }
}
